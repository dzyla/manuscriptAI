**Goal:** Replace six sidebar tabs and a 53-control toolbar with a page-first shell: a four-task rail, a selection-aware top bar, one position-ordered review list, and a page canvas bound to the document's real geometry.

## Review Focus

Five conditions the spec implies that no task's own happy path exercises. Each has its test added to the task that owns the code.

1. **A document with no `format`** (a row saved before A1's migration, or a fresh blank document). The canvas must fall back to the NIH preset rather than render a zero-width page or crash. — Task 1.
2. **A suggestion and a style flag covering the same span.** Both rows would offer to rewrite the same text; accepting one invalidates the other. The list must render both but disable the second's action once the first is applied. — Task 4.
3. **An empty document.** Page estimate, review list, rail badges and the status bar must all read sensibly at zero sentences rather than showing `0 of 12 pp` as if measured. — Tasks 1 and 4.
4. **A DOCX import landing while a panel is open.** A2a's import replaces the document wholesale; open panel state referencing the old document (a selected suggestion, a scrolled row) must reset rather than point at stale offsets. — Task 4.
5. **Zoom at other than 100%.** Page guides are absolutely positioned from measured offsets; they must stay aligned to the text when `editorZoom` scales the canvas. — Task 1.

```ts
// src/services/pagination.ts
import type { JSONContent } from '@tiptap/core';
import type { DocFormat } from '../types';
import { PAGE_SIZES_IN, NIH_PRESET } from './formatPresets';

/**
 * The ONLY source of a page number in the app. Both the page guides and the
 * status bar read this; nothing else computes one.
 *
 * Sub-project A3 replaces the body of `estimatePages` with real layout
 * measurement and flips `approximate` to false. No UI changes when it does:
 * the shape below is the contract. Until then every figure is a lower bound
 * that ignores figure and table flow, orphan control and section breaks.
 */
export interface PageEstimate {
  pages: number;
  /** true while this is a character-density estimate rather than real layout */
  approximate: boolean;
  /** character offsets after which a page boundary is estimated to fall */
  breaks: number[];
}

function fmt(format: DocFormat | null): DocFormat {
  return format ?? NIH_PRESET;
}

export function contentWidthIn(format: DocFormat | null): number {
  const f = fmt(format);
  const { width } = PAGE_SIZES_IN[f.page.size];
  return width - f.page.marginsIn.left - f.page.marginsIn.right;
}

function contentHeightIn(f: DocFormat): number {
  const { height } = PAGE_SIZES_IN[f.page.size];
  return height - f.page.marginsIn.top - f.page.marginsIn.bottom;
}

/** Characters per line at this width and point size, from the NIH 15-chars-per-inch guidance. */
function charsPerLine(f: DocFormat): number {
  const scale = 11 / Math.max(1, f.defaults.fontSizePt);
  return Math.max(20, Math.round(contentWidthIn(f) * 15 * scale));
}

/** Lines per page, from the NIH 6-lines-per-inch guidance. */
function linesPerPage(f: DocFormat): number {
  const scale = 11 / Math.max(1, f.defaults.fontSizePt);
  return Math.max(5, Math.round(contentHeightIn(f) * 6 * scale));
}

function walk(node: JSONContent, out: { text: string; hardBreaks: number[] }): void {
  if (node.type === 'pageBreak') { out.hardBreaks.push(out.text.length); return; }
  if (node.type === 'text') { out.text += node.text ?? ''; return; }
  for (const child of node.content ?? []) walk(child, out);
  // A block ends its line.
  if (node.type && node.type !== 'doc' && node.type !== 'text') out.text += '\n';
}

export function estimatePages(doc: JSONContent, format: DocFormat | null): PageEstimate {
  const f = fmt(format);
  const acc = { text: '', hardBreaks: [] as number[] };
  walk(doc, acc);

  const cpl = charsPerLine(f);
  const lpp = linesPerPage(f);
  const perPage = cpl * lpp;

  const breaks: number[] = [];
  let consumed = 0;
  for (const at of acc.hardBreaks) {
    // A hard break ends the current page wherever it falls.
    while (at - consumed > perPage) { consumed += perPage; breaks.push(consumed); }
    consumed = at;
    breaks.push(at);
  }
  while (acc.text.length - consumed > perPage) { consumed += perPage; breaks.push(consumed); }

  return { pages: Math.max(1, breaks.length + 1), approximate: true, breaks };
}
```

export type RailTask = 'write' | 'cite' | 'review' | 'format';
export interface RailBadges { write?: number; cite?: number; review?: number; format?: number }

const TASKS: { key: RailTask; label: string; hint: string; Icon: typeof Pencil }[] = [
  { key: 'write',  label: 'Write',  hint: 'Outline, history and versions',        Icon: Pencil },
  { key: 'cite',   label: 'Cite',   hint: 'Sources, search and references',       Icon: Quote },
  { key: 'review', label: 'Review', hint: 'Suggestions, style and chat',          Icon: CircleDot },
  { key: 'format', label: 'Format', hint: 'Styles, page setup and grant template', Icon: Pilcrow },
];

/**
 * One panel is open at a time. Selecting the active task again is how the user
 * closes the panel, so `onSelect` fires for it too and the caller decides.
 */
export function Rail({ active, badges, onSelect }: {
  active: RailTask | null;
  badges: RailBadges;
  onSelect: (t: RailTask) => void;
}) {
  return (
    <nav className="u-rail" aria-label="Tasks">
      {TASKS.map(({ key, label, hint, Icon }) => {
        const count = badges[key];
        return (
          <button
            key={key}
            type="button"
            aria-pressed={active === key}
            aria-label={`${label} — ${hint}`}
            title={`${label} — ${hint}`}
            className={active === key ? 'u-rail-btn is-active' : 'u-rail-btn'}
            onClick={() => onSelect(key)}
          >
            <Icon size={16} />
            {count ? <span className="u-rail-badge" data-testid={`badge-${key}`}>{count}</span> : null}
          </button>
        );
      })}
    </nav>
  );
}

// src/components/shell/StatusBar.tsx
import type { PageEstimate } from '../../services/pagination';

const APPROX_NOTE = 'Estimated from character density. It ignores figure and table flow, so treat it as a lower bound until pagination lands.';

export function StatusBar({ estimate, limitPages, words, voiceDistance, saveState }: {
  estimate: PageEstimate;
  limitPages: number | null;
  words: number;
  voiceDistance: number | null;
  saveState: string;
}) {
  const n = `${estimate.approximate ? '~' : ''}${estimate.pages}`;
  const pages = limitPages ? `${n} of ${limitPages}` : `${n} pp`;
  const over = limitPages !== null && estimate.pages > limitPages;
  return (
    <div className="u-status" role="status">
      <span
        className={over ? 'u-status-pages is-over' : 'u-status-pages'}
        title={estimate.approximate ? APPROX_NOTE : undefined}
      >
        {pages}
      </span>
      <span>{words.toLocaleString()} words</span>
      {voiceDistance !== null ? <span>voice {voiceDistance}</span> : null}
      <span className="u-status-save">{saveState}</span>
    </div>
  );
}

// src/components/shell/resolveEntryPoint.tsx
import { render, fireEvent, screen, cleanup } from '@testing-library/react';
import type { EntryPoint } from './reachability.test';
import App from '../../App';

/**
 * Walks an entry point's path against a rendered App and resolves true when the
 * final step is present and enabled. Test-only: imported by reachability.test.tsx.
 */
export async function resolveEntryPoint(entry: EntryPoint): Promise<boolean> {
  cleanup();
  render(<App />);
  for (const step of entry.path) {
    const [kind, id] = step.split(':');
    const ok = await visit(kind, id, entry);
    if (!ok) throw new Error(`"${entry.feature}" is unreachable: step "${step}" not found`);
  }
  return true;
}

async function visit(kind: string, id: string, entry: EntryPoint): Promise<boolean> {
  switch (kind) {
    case 'rail':   return click(screen.queryByRole('button', { name: new RegExp(`^${id}`, 'i') }));
    case 'topbar': return click(screen.queryByTestId(`topbar-${id}`));
    case 'section':return click(screen.queryByTestId(`section-${id}`));
    case 'row':    return !!screen.queryByTestId(`row-${id}`);
    case 'item':
    case 'control': {
      const el = screen.queryByTestId(`${kind === 'item' ? 'item' : 'control'}-${id}`);
      if (!el) return false;
      // A control that requires a selection may be disabled; present-and-labelled is enough.
      if (entry.needsSelection) return true;
      return !(el as HTMLButtonElement).disabled;
    }
    case 'figure': return !!screen.queryByTestId('figure-selected');
    default: return false;
  }
}

function click(el: HTMLElement | null): boolean {
  if (!el) return false;
  fireEvent.click(el);
  return true;
}

```ts
// src/components/panels/reviewRows.ts
import type { Suggestion, SentenceAudit } from '../../types';

export type ReviewKind = 'suggestion' | 'style' | 'grammar';

export interface ReviewRow {
  id: string;
  kind: ReviewKind;
  start: number;
  end: number;
  title: string;
  /** H's evidence string, verbatim. Never paraphrase: four fix rounds made these truthful. */
  evidence: string;
  canAct: boolean;
  actions: ('accept' | 'reject' | 'rewrite')[];
}

function tellPhrase(tells: string[], words: number): string {
  if (tells.length) return `${words} words · ${tells.join(', ')}`;
  return `${words} words · no specific pattern`;
}

export function reviewRows(args: {
  suggestions: Suggestion[];
  flagged: SentenceAudit[];
  grammar: { start: number; end: number; message: string }[];
  plainText: string;
  appliedIds: Set<string>;
}): ReviewRow[] {
  const { suggestions, flagged, grammar, plainText, appliedIds } = args;

  // Spans already rewritten by an applied suggestion: anything else pointing here is stale.
  const changed: { start: number; end: number }[] = suggestions
    .filter(s => appliedIds.has(s.id))
    .map(s => ({ start: s.startIndex, end: s.endIndex }));
  const overlapsChanged = (start: number, end: number) =>
    changed.some(c => start < c.end && end > c.start);

  const rows: ReviewRow[] = [];

  for (const s of suggestions) {
    if (appliedIds.has(s.id)) continue;
    const intact = plainText.slice(s.startIndex, s.endIndex) === s.originalText;
    rows.push({
      id: `suggestion:${s.id}`,
      kind: 'suggestion',
      start: s.startIndex,
      end: s.endIndex,
      title: s.agent,
      evidence: intact ? s.explanation : `${s.explanation} (the text here has already changed)`,
      canAct: intact && !overlapsChanged(s.startIndex, s.endIndex),
      actions: s.suggestedText ? ['accept', 'reject'] : ['reject'],
    });
  }

  for (const f of flagged) {
    const stale = overlapsChanged(f.start, f.end);
    rows.push({
      id: `style:${f.start}-${f.end}`,
      kind: 'style',
      start: f.start,
      end: f.end,
      title: tellPhrase(f.tells, f.words),
      evidence: stale
        ? `${tellPhrase(f.tells, f.words)} — this sentence has already changed since it was measured`
        : tellPhrase(f.tells, f.words),
      canAct: !stale,
      actions: ['rewrite'],
    });
  }

  for (const g of grammar) {
    rows.push({
      id: `grammar:${g.start}-${g.end}`,
      kind: 'grammar',
      start: g.start,
      end: g.end,
      title: 'Grammar',
      evidence: g.message,
      canAct: !overlapsChanged(g.start, g.end),
      actions: ['accept', 'reject'],
    });
  }

  return rows.sort((a, b) => a.start - b.start || a.kind.localeCompare(b.kind));
}
```
